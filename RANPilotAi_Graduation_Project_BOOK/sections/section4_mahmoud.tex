% ── Slide 30 — Overview & Motivation
\begin{frame}{RAG Assistant --- Overview \& Motivation}
  \begin{columns}[T]
    \column{.45\textwidth}
      \begin{block}{The Problem}
        \begin{itemize}
          \item Engineers waste hours manually searching 3GPP PDF specs
          \item Keyword search fails when wording differs from the spec
          \item General LLMs \textbf{hallucinate} protocol details
          \item No tool provides citation-backed, spec-grounded answers
        \end{itemize}
      \end{block}
    \column{.45\textwidth}
      \begin{block}{Our Solution}
        \begin{itemize}
          \item \textbf{Engineering-grade RAG} purpose-built for 3GPP
          \item Fully \textbf{local}: embeddings, vector store, re-ranking
          \item Every answer cites: TS number, Release, Section, page
          \item Integrated into RANPilotAI --- no tool switching
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

% ── Slide 31b — Pipeline key engineering decisions
\begin{frame}{Pipeline --- Key Engineering Decisions}
  \small
  \begin{columns}[T]
    \column{.50\textwidth}
      \textbf{Ingestion \& Chunking}
      \begin{itemize}
        \item \textbf{Section-aware} PDF parsing (PyMuPDF) splits on
              3GPP section boundaries --- formulas and ASN.1 never cut mid-way
        \item Target chunk: 200--350 tokens (max 500)
        \item 8 chunk types: section, event\_def, ASN.1,
              table, figure, annex, history, formula
      \end{itemize}
    \column{.50\textwidth}
      \textbf{Retrieval \& Generation}
      \begin{itemize}
        \item \textbf{BGE-large-en-v1.5} embedder
              $\rightarrow$ local \textbf{Qdrant} vector store
        \item Top-20 retrieved $\rightarrow$ cross-encoder reranker
              $\rightarrow$ top-5 sent to LLM
        \item \textbf{Anti-hallucination prompt}: model must say
              \emph{``cannot find in specifications''} rather than guess
        \item Metadata filter: restrict to specific TS / Release
      \end{itemize}
  \end{columns}
\end{frame}
